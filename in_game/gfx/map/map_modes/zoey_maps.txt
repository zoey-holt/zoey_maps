population_capacity = {
	map_color = {
		if = {
			limit = { is_land = no }
            value = terrain_ocean
		}
		else_if = {
			limit = { is_ownable = no }
            value = define:NMapColors|MAP_COLOR_MIN
		}
		else_if = {
			limit = {
                location_max_population < 50
            }
			value = define:NMapColors|MAP_COLOR_LOW
		}
		else_if = {
			limit = {
                location_max_population < 250
            }
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_MID
				factor = {
                    value = location_max_population
					add = -50
                    divide = 200
                }
			}
		}
		else_if = {
			limit = {
                location_max_population < 1000
            }
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_MID
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = {
                    value = location_max_population
					add = -250
                    divide = 750
                }
			}
		}
		else = {
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_HIGH
				max_color = define:NMapColors|MAP_COLOR_TOP
				factor = {
                    value = location_max_population
					add = -1000
                    divide = 1000
                }
			}
		}
	}
	secondary_map_color = {
		if = {
			limit = {
                location_population_percentage > 1
            }
            value = define:NMapColors|MAP_COLOR_MIN
		}
	}
	tooltip_key = MAPMODE_POPULATION_CAPACITY_TT

	small_map_names = location
	medium_map_names = none
	large_map_names = none

	large_tooltip_context = location

	use_fow = no

	category = population
	index = 0

		market_marker = no
		fort_marker = no
		toll_marker = no
	}

		zoom_step = @zoom_step_near

		gradient_alpha_inside = 	@data_gradient_alpha_inside
		gradient_alpha_outside = 	@data_gradient_alpha_outside
		gradient_width = 			@data_gradient_width
		gradient_color_mult =		@data_gradient_color_mult

			# 100% = yellow
			# 150% = red
			limit = {
				location_population_percentage < 1.5
            }
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_MID
				max_color = define:NMapColors|MAP_COLOR_LOW
				factor = {
                    value = location_population_percentage
                    add = -1
                    multiply = 2
                }
			}
		}
		else = {
			# 150% = red
			# 1000% = magenta (probably only applicable at game start)
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = rgb { 128 0 128 }
				factor = {
                    value = location_population_percentage
                    add = -1.5
					divide = 10
                }
			}
		}
	}
	secondary_map_color = {
		if = {
			limit = {
